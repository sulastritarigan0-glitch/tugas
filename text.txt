/* ================= UTIL ================= */

function rupiah(number){

    return "Rp" +
    number.toLocaleString("id-ID");

}

function show(id){

    document
    .getElementById(id)
    .classList.remove("hidden");

}

function hide(id){

    document
    .getElementById(id)
    .classList.add("hidden");

}


/* ================= PRODUCTS ================= */

function renderProducts(){

    const search =
    document
    .getElementById("search")
    .value
    .toLowerCase();

    const list =
    products
    .map((p,index)=>({p,index}))
    .filter(x=>

        (
            x.p[0]+" "+
            x.p[1]
        )
        .toLowerCase()
        .includes(search)

    );

    document
    .getElementById("products")
    .innerHTML =

    list.map(x=>`

        <div class="card">

            <img
            src="${x.p[3]}"
            alt="${x.p[0]}">

            <h3>
            ${x.p[0]}
            </h3>

            <span class="badge">
            ${x.p[1]}
            </span>

            <div class="price">
            ${rupiah(x.p[2])}
            </div>

            <button
            class="primary"
            onclick="addCart(${x.index})">

            🛒 Tambahkan

            </button>

        </div>

    `).join("");

}


/* ================= CART ================= */

function addCart(index){

    const product =
    products[index];

    const existing =
    cart.find(
        x=>x.name===product[0]
    );

    if(existing){

        existing.qty++;

    }else{

        cart.push({

            name:product[0],

            price:product[2],

            qty:1

        });

    }

    renderCart();

}


function renderCart(){

    let total = 0;

    let count = 0;

    const box =
    document
    .getElementById("cartItems");


    if(cart.length===0){

        box.innerHTML =
        `<p style="color:#aaa">
        Keranjang masih kosong.
        </p>`;

    }else{

        box.innerHTML =

        cart.map((item,index)=>{

            total +=
            item.price *
            item.qty;

            count +=
            item.qty;

            return `

            <div class="cart-row">

                <div>

                    <b>
                    ${item.name}
                    </b>

                    <br>

                    ${rupiah(item.price)}

                </div>

                <div>

                    <button
                    onclick="minus(${index})">

                    −

                    </button>

                    ${item.qty}

                    <button
                    onclick="plus(${index})">

                    +

                    </button>

                </div>

            </div>

            `;

        }).join("");

    }


    document
    .getElementById("total")
    .textContent =
    rupiah(total);

    document
    .getElementById("cartCount")
    .textContent =
    count;

}


function plus(index){

    cart[index].qty++;

    renderCart();

}


function minus(index){

    cart[index].qty--;

    if(cart[index].qty<=0){

        cart.splice(index,1);

    }

    renderCart();

}


function goCart(){

}

function previewProof(event){

    const file =
    event.target.files[0];

    if(!file){

        return;

    }

    if(
        !file.type.startsWith("image/")
    ){

        alert(
            "Pilih file gambar."
        );

        return;

    }


    const reader =
    new FileReader();


    reader.onload=function(e){

        document
        .getElementById("proofImage")
        .src =
        e.target.result;

        document
        .getElementById("proofPreview")
        .style.display =
        "block";

        proofUploaded = true;

    };


    reader.readAsDataURL(file);

}

    .getElementById("address")
    .scrollIntoView({

    });

}


/* ================= ORDER ================= */

function finishOrder(){

    const name =
    document
    .getElementById("customerName")
    .value.trim();


    const phone =
    document
    .getElementById("phone")
    .value.trim();


    const address =
    document
    .getElementById("addressText")
    .value.trim();


    const city =
    document
    .getElementById("city")
    .value.trim();


    if(
        !name ||
        !phone ||
        !address ||
        !city
    ){

        alert(
            "Lengkapi data alamat terlebih dahulu."
        );

        return;

    }


    const total =
    cart.reduce(
        (sum,item)=>
        sum +
        item.price *
        item.qty,
        0
    );


    const orderNumber =
    "BUIDA-" +
    Math.floor(
        100000 +
        Math.random()*900000
    );


    document
    .getElementById("orderNumber")
    .textContent =
    orderNumber;


    document
    .getElementById("receiptName")
    .textContent =
    name;


    document
    .getElementById("receiptAddress")
    .textContent =
    address;


    document
    .getElementById("receiptCity")
    .textContent =
    city;


    document
    .getElementById("receiptPayment")
    .textContent =
    selectedPayment;


    document
    .getElementById("receiptTotal")
    .textContent =
    rupiah(total);


    show("receipt");

    show("tracking");

    show("chat");


    document
    .getElementById("tracking")
    .scrollIntoView({
        behavior:"smooth"
    });


    startTracking();

}


/* ================= AUTOMATIC TRACKING ================= */

function startTracking(){

    if(trackingStarted){

        return;

    }

    trackingStarted = true;


    const steps =
    document
    .querySelectorAll(".step");


    const messages = [

        "Pesanan sedang diproses oleh toko.",

        "Kurir sudah mengambil paket.",

        "Paket sedang menuju kota tujuan.",

        "Paket sudah sampai di alamat penerima."

    ];


    const labels = [

        "Pesanan sedang diproses.",

        "Kurir sedang mengambil paket.",

        "Kurir sedang dalam perjalanan.",

        "Paket sudah sampai."

    ];


    const positions = [

        "28%",

        "48%",

        "70%",

        "88%"

    ];


    let current = 1;


    const timer =
    setInterval(()=>{

        steps[current]
        .classList
        .add("active");


        document
        .getElementById(
            "step"+current
        )
        .textContent =
        messages[current-1];


        document
        .getElementById(
            "mapLabel"
        )
        .textContent =
        labels[current-1];


        document
        .getElementById(
            "courierDot"
        )
        .style.left =
        positions[current-1];


        current++;


        if(current>4){

            clearInterval(timer);

        }


    },5000);

}


/* ================= CHAT ================= */

function addMessage(
    text,
    sender,
    image=null
){

    const chat =
    document
    .getElementById("chatArea");


    const message =
    document
    .createElement("div");


    message.className =
    "message " +
    (
        sender==="me"
        ? "me"
        : "them"
    );


    const name =
    document
    .createElement("b");


    name.textContent =
    sender==="me"
    ? "Kamu"
    : "Raka";


    message.appendChild(name);


    if(image){

        const img =
        document
        .createElement("img");

        img.src = image;

        message.appendChild(img);

    }else{

        const p =
        document
        .createElement("p");

        p.textContent =
        text;

        message.appendChild(p);

    }


    const time =
    document
    .createElement("small");


    time.textContent =
    new Date()
    .toLocaleTimeString(
        "id-ID",
        {
            hour:"2-digit",
            minute:"2-digit"
        }
    )
    + " ✓✓";


    message.appendChild(time);


    chat.appendChild(message);


    chat.scrollTop =
    chat.scrollHeight;

}


function sendMessage(){

    const input =
    document
    .getElementById("chatInput");


    const text =
    input.value.trim();


    if(!text){

        return;

    }


    addMessage(
        text,
        "me"
    );


    input.value="";


    const typing =
    document
    .getElementById("typing");


    typing.style.display =
    "block";


    setTimeout(()=>{

        typing.style.display =
        "none";


        const lower =
        text.toLowerCase();


        let reply =
        "Baik, pesan kamu sudah saya terima.";


        if(
            lower.includes("halo") ||
            lower.includes("hai")
        ){

            reply =
            "Halo! Saya Raka, pengantar dari Bu Ida Store.";

        }


        else if(
            lower.includes("dimana") ||
            lower.includes("di mana")
        ){

            reply =
            "Paketmu sedang dalam perjalanan menuju kota tujuan.";

        }


        else if(
            lower.includes("kapan")
        ){

            reply =
            "Perkiraan paket sampai hari ini dalam simulasi.";

        }


        else if(
            lower.includes("terima kasih") ||
            lower.includes("makasih")
        ){

            reply =
            "Sama-sama! Semoga paketnya sampai dengan aman.";

        }


        addMessage(
            reply,
            "them"
        );


    },1200);

}


/* ================= SEND PHOTO ================= */

function sendPhoto(event){

    const file =
    event.target.files[0];


    if(!file){

        return;

    }


    if(
        !file.type.startsWith("image/")
    ){

        alert(
            "Pilih file foto."
        );

        return;

    }


    const reader =
    new FileReader();


    reader.onload=function(e){

        addMessage(
            "",
            "me",
            e.target.result
        );


        setTimeout(()=>{

            addMessage(
                "Foto sudah saya terima.",
                "them"
            );

        },900);

    };


    reader.readAsDataURL(file);

}


/* ================= CALL SIMULATION ================= */

let callTimer;


function startCall(type){

    const screen =
    document
    .getElementById("callScreen");


    const status =
    document
    .getElementById("callStatus");


    screen.style.display =
    "flex";


    status.textContent =
    (
        type==="video"
        ? "📹 Video call"
        : "📞 Panggilan suara"
    )
    +
    " • Menghubungkan...";


    clearTimeout(callTimer);


    callTimer =
    setTimeout(()=>{

        status.textContent =
        (
            type==="video"
            ? "📹 Video call"
            : "📞 Panggilan suara"
        )
        +
        " • Aktif • Simulasi";

    },1500);

}


function endCall(){

    document
    .getElementById("callScreen")
    .style.display =
    "none";


    clearTimeout(callTimer);

}
